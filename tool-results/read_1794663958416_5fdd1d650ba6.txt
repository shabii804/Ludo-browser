   187→function schedule(ms: number, fn: () => void) {
   188→  if (pendingTimer) clearTimeout(pendingTimer)
   189→  pendingTimer = setTimeout(fn, ms)
   190→}

   192→function cancelSchedule() {
   193→  if (pendingTimer) clearTimeout(pendingTimer)
   194→  pendingTimer = null
   195→}

   197→/** Walk of intermediate positions for a move (for step animation). */
   198→function movePath(game: GameState, move: Move): Array<{ location: 'track' | 'home' | 'finished'; trackIdx: number; homeStep: number }> {
   199→  const color = game.players[game.turn].color
   200→  const token = game.players[game.turn].tokens[move.tokenId]
   201→  const die = game.dice[move.dieIndex]
   202→  const from = token.location === 'yard' ? -1 : token.location === 'home' ? 50 + token.homeStep : token.location === 'finished' ? 56 : token.progress
   203→  const to = token.location === 'yard' ? 0 : Math.min(from + die, 56)
   204→  const steps: Array<{ location: 'track' | 'home' | 'finished'; trackIdx: number; homeStep: number }> = []
   205→  for (let p = from + 1; p <= to; p++) {
   206→    if (p <= 50) steps.push({ location: 'track', trackIdx: trackIdxFor(color, p), homeStep: 0 })
   207→    else if (p < 56) steps.push({ location: 'home', trackIdx: -1, homeStep: p - 50 })
   208→    else steps.push({ location: 'finished', trackIdx: -1, homeStep: 0 })
   209→  }
   210→  return steps
   211→}
   212→
   213→function patchToken(game: GameState, move: Move, step: { location: string; trackIdx: number; homeStep: number }): GameState {
   214→  const pIdx = game.turn
   215→  return {
   216→    ...game,
   217→    players: game.players.map((p, i) =>
   218→      i !== pIdx
   219→        ? p
   220→        : {
   221→            ...p,
   222→            tokens: p.tokens.map((t) =>
   223→              t.id === move.tokenId ? { ...t, location: step.location as Token['location'], trackIdx: step.trackIdx, homeStep: step.homeStep } : t
   224→            ),
   225→          }
   226→    ),
   227→  }
   228→}

   240→export const useLudo = create<LudoStore>((set, get) => {
   241→  // ---- core flow helpers (defined inside create for set/get access) ----
   242→
   243→  function pushFloat(text: string, kind: FloatMsg['kind'] = 'info') {
   244→    const id = msgId++
   245→    set((s) => ({ floatMsgs: [...s.floatMsgs, { id, text, kind }].slice(-4) }))
   246→    schedule(1800, () => get().clearFloat(id))
   247→  }
   248→
   249→  function startTurnTimer() {
   250→    set({ turnDeadline: Date.now() + TURN_SECONDS * 1000 })
   251→  }
   252→
   253→  function finishGame(result: GameState) {
   254→    const session = get().session
   255→    if (!session || !result.winner) return
   256→    const winnerPlayer = result.players.find((p) => p.color === result.winner)
   257→    const youWon = result.winner === session.youColor
   258→    const pot = session.pot
   259→    const coinDelta = youWon ? pot : 0
   260→    if (youWon) {
   261→      get().addCoins(pot)
   262→      sfx.win()
   263→      set((s) => ({ profile: { ...s.profile, wins: s.profile.wins + 1 } }))
   264→      setTimeout(() => get().updateProfile({}), 0)
   265→    } else {
   266→      sfx.lose()
   267→      set((s) => ({ profile: { ...s.profile, losses: s.profile.losses + 1 } }))
   268→      setTimeout(() => get().updateProfile({}), 0)
   269→    }
   270→    cancelSchedule()
   271→    set({
   272→      winData: { winner: result.winner, winnerName: winnerPlayer?.name ?? result.winner, pot, youWon, coinDelta },
   273→      animating: false,
   274→      canRoll: false,
   275→      movableTokenIds: [],
   276→      turnDeadline: null,
   277→    })
   278→    saveProfile(get().profile)
   279→  }
   280→
   281→  /** After every applied move / pass: decide who acts next (local games only). */
   282→  function scheduleNext() {
   283→    const s = get()
   284→    if (!s.game || s.game.phase === 'gameover' || s.session?.online) return
   285→    const gen = ++loopGen
   286→    const g = s.game
   287→    const cur = g.players[g.turn]
   288→
   289→    if (g.phase === 'roll') {
   290→      if (cur.isBot) {
   291→        schedule(700, () => void botRoll(gen))
   292→      } else {
   293→        set({ canRoll: true, movableTokenIds: [], selectedDie: 0, turnDeadline: null })
   294→        startTurnTimer()
   295→      }
   296→      return
   297→    }
   298→
   299→    if (g.phase === 'move') {
   300→      const moves = legalMoves(g)
   301→      if (moves.length === 0) {
   302→        pushFloat(cur.isBot ? `${cur.name} has no move` : 'No possible move', 'pass')
   303→        sfx.pass()
   304→        const passed = passTurn(g)
   305→        set({ game: passed })
   306→        schedule(900, () => scheduleNext())
   307→        return
   308→      }
   309→      if (cur.isBot) {
   310→        schedule(600, () => void botMove(gen))
   311→        return
   312→      }
   313→      // human: compute movable tokens for selected die; auto-switch die if needed
   314→      set({ canRoll: false })
   315→      computeMovable()
   316→    }
   317→  }
   318→
   319→  function computeMovable() {
   320→    const g = get().game
   321→    if (!g || g.phase !== 'move') return
   322→    let dieIdx = get().selectedDie
   323→    let moves = legalMoves(g).filter((m) => m.dieIndex === dieIdx)
   324→    if (moves.length === 0) {
   325→      // auto-switch to a die that has moves
   326→      const all = legalMoves(g)
   327→      if (all.length > 0) {
   328→        dieIdx = all[0].dieIndex
   329→        set({ selectedDie: dieIdx })
   330→        moves = all.filter((m) => m.dieIndex === dieIdx)
   331→      }
   332→    }
   333→    set({ movableTokenIds: [...new Set(moves.map((m) => m.tokenId))] })
   334→  }
   335→
   336→  async function animateAndApply(g0: GameState, move: Move) {
   337→    const gen = loopGen
   338→    set({ animating: true, movableTokenIds: [], canRoll: false, turnDeadline: null })
   339→    const steps = movePath(g0, move)
   340→    for (const step of steps) {
   341→      if (gen !== loopGen) return
   342→      await sleep(step.location === 'finished' ? 260 : 130)
   343→      if (gen !== loopGen) return
   344→      set({ game: patchToken(g0, move, step) })
   345→      sfx.step()
   346→    }
   347→    if (gen !== loopGen) return
   348→    await sleep(120)
   349→    const result = applyMove(g0, move)
   350→    if (move.captures.length > 0) {
   351→      sfx.capture()
   352→      pushFloat(`Captured ${move.captures.length} token${move.captures.length > 1 ? 's' : ''}!`, 'capture')
   353→    }
   354→    if (move.finishes) sfx.finish()
   355→    set({ game: result, animating: false, diceValues: [] })
   356→    if (result.phase === 'gameover' && result.winner) {
   357→      finishGame(result)
   358→      return
   359→    }
   360→    scheduleNext()
   361→  }
   362→
   363→  async function botRoll(gen: number) {
   364→    if (gen !== loopGen) return
   365→    const g = get().game
   366→    if (!g || g.phase !== 'roll') return
   367→    set({ canRoll: false, diceRolling: true })
   368→    sfx.dice()
   369→    await sleep(650)
   370→    if (gen !== loopGen) return
   371→    const { state, rolled, forfeited } = rollDice(g)
   372→    set({ game: state, diceValues: rolled, diceRolling: false, selectedDie: 0 })
   373→    if (rolled[0] === 6 || (state.mode === 'master' && rolled.every((d) => d === 6))) sfx.six()
   374→    if (forfeited) {
   375→      pushFloat(`Three sixes! ${g.players[g.turn].name} loses the turn`, 'forfeit')
   376→      const passed = passTurn(state)
   377→      set({ game: passed })
   378→      schedule(1100, () => scheduleNext())
   379→      return
   380→    }
   381→    await sleep(250)
   382→    if (gen !== loopGen) return
   383→    scheduleNext()
   384→  }
   385→
   386→  async function botMove(gen: number) {
   387→    if (gen !== loopGen) return
   388→    const g = get().game
   389→    if (!g || g.phase !== 'move') return
   390→    const move = botChooseMove(g)
   391→    if (!move) {
   392→      const passed = passTurn(g)
   393→      set({ game: passed })
   394→      schedule(900, () => scheduleNext())
   395→      return
   396→    }
   397→    set({ selectedDie: move.dieIndex, diceValues: [...g.dice] })
   398→    await animateAndApply(g, move)
   399→  }

   403→  function wireOnline() {
   404→    const sock = getSocket()
   405→    sock.off('connect')
   406→    sock.off('disconnect')
   407→    sock.on('connect', () => {
   408→      set((s) => ({ onlineStatus: s.lobby ? 'lobby' : 'connected', onlineError: '' }))
   409→    })
   410→    sock.on('disconnect', () => {
   411→      set((s) => (s.onlineStatus === 'playing' ? { onlineStatus: 'playing' } : { onlineStatus: 'idle' }))
   412→    })
   413→    online.on('lobby:created', ({ code }: { code: string }) => {
   414→      set({ onlineStatus: 'lobby' })
   415→    })
   416→    online.on('lobby:update', ({ room }: { room: LobbyInfo }) => {
   417→      set({ onlineStatus: 'lobby', lobby: room, onlineError: '' })
   418→    })
   419→    online.on('lobby:error', ({ message }: { message: string }) => {
   420→      set({ onlineError: message })
   421→    })
   422→    online.on('game:started', ({ state, yourColor, roomCode, stake }: { state: GameState; yourColor: Color; roomCode: string; stake: number }) => {
   423→      loopGen++
   424→      cancelSchedule()
   425→      const pot = stake * state.players.length
   426→      const me = get().profile
   427→      set({
   428→        view: 'game',
   429→        session: { mode: state.mode, stake, online: true, pot, youColor: yourColor, roomCode },
   430→        game: state,
   431→        onlineStatus: 'playing',
   432→        winData: null,
   433→        animating: false,
   434→        diceRolling: false,
   435→        diceValues: [],
   436→        canRoll: false,
   437→        movableTokenIds: [],
   438→        selectedDie: 0,
   439→        chat: [],
   440→        floatMsgs: [],
   441→        rematchVotes: [],
   442→        lobby: null,
   443→        turnDeadline: null,
   444→      })
   445→      get().addCoins(-stake) // pay entry
   446→      sfx.coin()
   447→      void me
   448→      // drive my turn if it starts now
   449→      setTimeout(() => scheduleNextOnline(), 400)
   450→    })
   451→    online.on('game:rolled', ({ state, rolled, forfeited }: { state: GameState; rolled: number[]; forfeited: boolean }) => {
   452→      const gen = ++loopGen
   453→      cancelSchedule()
   454→      set({ game: state, diceValues: rolled, diceRolling: false, canRoll: false, selectedDie: 0 })
   455→      if (rolled[0] === 6) sfx.six()
   456→      if (forfeited) {
   457→        pushFloat('Three sixes! Turn forfeited', 'forfeit')
   458→      }
   459→      setTimeout(() => scheduleNextOnline(), 300)
   460→      void gen
   461→    })
   462→    online.on('game:moved', ({ state, move }: { state: GameState; move: OnlineMove }) => {
   463→      // replay animation from previous visible state, then snap to authoritative state
   464→      const g0 = get().game
   465→      const gen = ++loopGen
   466→      cancelSchedule()
   467→      if (g0 && move) {
   468→        void (async () => {
   469→          set({ animating: true, movableTokenIds: [] })
   470→          const steps = movePathForOnline(g0, move)
   471→          for (const step of steps) {
   472→            if (gen !== loopGen) return
   473→            await sleep(step.location === 'finished' ? 260 : 130)
   474→            if (gen !== loopGen) return
   475→            set({ game: patchTokenOnline(g0, move, step) })
   476→            sfx.step()
   477→          }
   478→          if (gen !== loopGen) return
   479→          if (move.captures.length > 0) {
   480→            sfx.capture()
   481→            pushFloat('Capture!', 'capture')
   482→          }
   483→          if (move.finishes) sfx.finish()
   484→          set({ game: state, animating: false, diceValues: [] })
   485→          setTimeout(() => scheduleNextOnline(), 250)
   486→        })()
   487→      } else {
   488→        set({ game: state, animating: false })
   489→        setTimeout(() => scheduleNextOnline(), 250)
   490→      }
   491→    })
   492→    online.on('game:passed', ({ state, reason }: { state: GameState; reason?: string }) => {
   493→      set({ game: state, diceValues: [] })
   494→      pushFloat(reason === 'forfeit' ? 'Three sixes! Turn forfeited' : 'No possible move', reason === 'forfeit' ? 'forfeit' : 'pass')
   495→      sfx.pass()
   496→      setTimeout(() => scheduleNextOnline(), 300)
   497→    })
   498→    online.on('game:state', ({ state }: { state: GameState }) => {
   499→      // full authoritative sync (bot takeover, reconnect)
   500→      const gen = ++loopGen
   501→      cancelSchedule()
   502→      set({ game: state, animating: false, diceRolling: false, diceValues: [], movableTokenIds: [] })
   503→      void gen
   504→      setTimeout(() => scheduleNextOnline(), 300)
   505→    })
   506→    online.on('game:over', ({ winner, winnerName, pot }: { winner: Color; winnerName: string; pot: number }) => {
   507→      const session = get().session
   508→      if (!session) return
   509→      const youWon = winner === session.youColor
   510→      if (youWon) {
   511→        get().addCoins(pot)
   512→        sfx.win()
   513→        set((s) => ({ profile: { ...s.profile, wins: s.profile.wins + 1 } }))
   514→      } else {
   515→        sfx.lose()
   516→        set((s) => ({ profile: { ...s.profile, losses: s.profile.losses + 1 } }))
   517→      }
   518→      saveProfile(get().profile)
   519→      set({
   520→        winData: { winner, winnerName, pot, youWon, coinDelta: youWon ? pot : 0 },
   521→        canRoll: false,
   522→        movableTokenIds: [],
   523→        animating: false,
   524→        turnDeadline: null,
   525→      })
   526→    })
   527→    online.on('game:chat', (msg: { color: Color; name: string; text: string }) => {
   528→      set((s) => ({ chat: [...s.chat, { id: msgId++, color: msg.color, name: msg.name, text: msg.text, ts: Date.now() }].slice(-50) }))
   529→    })
   530→    online.on('game:emote', ({ color, emoji }: { color: Color; emoji: string }) => {
   531→      sfx.emote()
   532→      set((s) => ({ emotes: [...s.emotes, { id: msgId++, color, emoji }].slice(-6) }))
   533→    })
   534→    online.on('game:rematch', ({ votes }: { votes: string[] }) => {
   535→      set({ rematchVotes: votes })
   536→      if (votes.length > 0) pushFloat('Rematch requested', 'info')
   537→    })
   538→    online.on('room:closed', ({ reason }: { reason: string }) => {
   539→      set({ onlineStatus: 'idle', lobby: null, session: null, game: null, view: 'menu', onlineError: reason || 'Room closed' })
   540→    })
   541→    online.on('connect_error', () => {
   542→      set({ onlineError: 'Cannot reach game server. Try again shortly.' })
   543→    })
   544→  }

   546→  /** Online equivalent of scheduleNext: only enables local input; bots are server-driven. */
   547→  function scheduleNextOnline() {
   548→    const s = get()
   549→    if (!s.game || s.game.phase === 'gameover' || !s.session?.online) return
   550→    const g = s.game
   551→    const cur = g.players[g.turn]
   552→    const isMe = cur.color === s.session.youColor && !cur.isBot
   553→    if (g.phase === 'roll') {
   554→      if (isMe) {
   555→        set({ canRoll: true })
   556→        startTurnTimer()
   557→      } else {
   558→        set({ canRoll: false })
   559→      }
   560→      return
   561→    }
   562→    if (g.phase === 'move') {
   563→      if (isMe) {
   564→        set({ canRoll: false })
   565→        computeMovable()
   566→      } else {
   567→        set({ movableTokenIds: [] })
   568→      }
   569→    }
   570→  }

   572→  function movePathForOnline(g0: GameState, move: OnlineMove) {
   573→    // find moving token in current display state (turn might differ from server snapshot ordering)
   574→    const color = move.color
   575→    const token = g0.players.find((p) => p.color === color)?.tokens[move.tokenId]
   576→    if (!token) return []
   577→    const die = move.die
   578→    const from = token.location === 'yard' ? -1 : token.location === 'home' ? 50 + token.homeStep : token.location === 'finished' ? 56 : token.progress
   579→    const to = token.location === 'yard' ? 0 : Math.min(from + die, 56)
   580→    const steps: Array<{ location: 'track' | 'home' | 'finished'; trackIdx: number; homeStep: number }> = []
   581→    for (let p = from + 1; p <= to; p++) {
   582→      if (p <= 50) steps.push({ location: 'track', trackIdx: trackIdxFor(color, p), homeStep: 0 })
   583→      else if (p < 56) steps.push({ location: 'home', trackIdx: -1, homeStep: p - 50 })
   584→      else steps.push({ location: 'finished', trackIdx: -1, homeStep: 0 })
   585→    }
   586→    return steps
   587→  }
