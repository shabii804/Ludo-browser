   239→function schedule(ms: number, fn: () => void) {

   249→/** Walk of intermediate positions for a move (for step animation). */
   250→function movePath(game: GameState, move: Move): Array<{ location: 'track' | 'home' | 'finished'; trackIdx: number; homeStep: number }> {
   251→  const color = game.players[game.turn].color
   252→  const token = game.players[game.turn].tokens[move.tokenId]
   253→  const die = game.dice[move.dieIndex]
   254→  const from = token.location === 'yard' ? -1 : token.location === 'home' ? 50 + token.homeStep : token.location === 'finished' ? 56 : token.progress
   255→  const to = token.location === 'yard' ? 0 : Math.min(from + die, 56)
   256→  const steps: Array<{ location: 'track' | 'home' | 'finished'; trackIdx: number; homeStep: number }> = []
   257→  for (let p = from + 1; p <= to; p++) {
   258→    if (p <= 50) steps.push({ location: 'track', trackIdx: trackIdxFor(color, p), homeStep: 0 })
   259→    else if (p < 56) steps.push({ location: 'home', trackIdx: -1, homeStep: p - 50 })
   260→    else steps.push({ location: 'finished', trackIdx: -1, homeStep: 0 })
   261→  }
   262→  return steps
   263→}

   265→function patchToken(game: GameState, move: Move, step: { location: string; trackIdx: number; homeStep: number }): GameState {
   266→  const pIdx = game.turn
   267→  return {
   268→    ...game,
   269→    players: game.players.map((p, i) =>
   270→      i !== pIdx
   271→        ? p
   272→        : {
   273→            ...p,
   274→            tokens: p.tokens.map((t) =>
   275→              t.id === move.tokenId ? { ...t, location: step.location as Token['location'], trackIdx: step.trackIdx, homeStep: step.homeStep } : t
   276→            ),
   277→          }
   278→    ),
   279→  }
   280→}

   292→export const useLudo = create<LudoStore>((set, get) => {
   293→  // ---- core flow helpers (defined inside create for set/get access) ----
   294→
   295→  function pushFloat(text: string, kind: FloatMsg['kind'] = 'info') {
   296→    const id = msgId++
   297→    set((s) => ({ floatMsgs: [...s.floatMsgs, { id, text, kind }].slice(-4) }))
   298→    schedule(1800, () => get().clearFloat(id))
   299→  }
   300→
   301→  function startTurnTimer() {
   302→    set({ turnDeadline: Date.now() + TURN_SECONDS * 1000 })
   303→  }
   304→
   305→  function finishGame(result: GameState) {
   306→    const session = get().session
   307→    if (!session || !result.winner) return
   308→    const winnerPlayer = result.players.find((p) => p.color === result.winner)
   309→    const youWon = result.winner === session.youColor
   310→    const pot = session.pot
   311→    const coinDelta = youWon ? pot : 0
   312→    if (youWon) {
   313→      get().addCoins(pot)
   314→      sfx.win()
   315→      set((s) => ({ profile: { ...s.profile, wins: s.profile.wins + 1, gems: s.profile.gems + 2 } }))
   316→      setTimeout(() => get().updateProfile({}), 0)
   317→    } else {
   318→      sfx.lose()
   319→      set((s) => ({ profile: { ...s.profile, losses: s.profile.losses + 1 } }))
   320→      setTimeout(() => get().updateProfile({}), 0)
   321→    }
   322→    cancelSchedule()
   323→    set({
   324→      winData: { winner: result.winner, winnerName: winnerPlayer?.name ?? result.winner, pot, youWon, coinDelta },
   325→      animating: false,
   326→      canRoll: false,
   327→      movableTokenIds: [],
   328→      turnDeadline: null,
   329→    })
   330→    saveProfile(get().profile)
   331→  }
   332→
   333→  /** After every applied move / pass: decide who acts next (local games only). */
   334→  function scheduleNext() {
   335→    const s = get()
   336→    if (!s.game || s.game.phase === 'gameover' || s.session?.online) return
   337→    const gen = ++loopGen
   338→    const g = s.game
   339→    const cur = g.players[g.turn]
   340→
   341→    if (g.phase === 'roll') {
   342→      if (cur.isBot) {
   343→        schedule(700, () => void botRoll(gen))
   344→      } else {
   345→        set({ canRoll: true, movableTokenIds: [], selectedDie: 0, turnDeadline: null })
   346→        startTurnTimer()
   347→      }
   348→      return
   349→    }
   350→
   351→    if (g.phase === 'move') {
   352→      const moves = legalMoves(g)
   353→      if (moves.length === 0) {
   354→        pushFloat(cur.isBot ? `${cur.name} has no move` : 'No possible move', 'pass')
   355→        sfx.pass()
   356→        const passed = passTurn(g)
   357→        set({ game: passed })
   358→        schedule(900, () => scheduleNext())
   359→        return
   360→      }
   361→      if (cur.isBot) {
   362→        schedule(600, () => void botMove(gen))
   363→        return
   364→      }
   365→      // human: compute movable tokens for selected die; auto-switch die if needed
   366→      set({ canRoll: false })
   367→      computeMovable()
   368→    }
   369→  }
   370→
   371→  function computeMovable() {
   372→    const g = get().game
   373→    if (!g || g.phase !== 'move') return
   374→    let dieIdx = get().selectedDie
   375→    let moves = legalMoves(g).filter((m) => m.dieIndex === dieIdx)
   376→    if (moves.length === 0) {
   377→      // auto-switch to a die that has moves
   378→      const all = legalMoves(g)
   379→      if (all.length > 0) {
   380→        dieIdx = all[0].dieIndex
   381→        set({ selectedDie: dieIdx })
   382→        moves = all.filter((m) => m.dieIndex === dieIdx)
   383→      }
   384→    }
   385→    set({ movableTokenIds: [...new Set(moves.map((m) => m.tokenId))] })
   386→  }
   387→
   388→  async function animateAndApply(g0: GameState, move: Move) {
   389→    const gen = loopGen
   390→    set({ animating: true, movableTokenIds: [], canRoll: false, turnDeadline: null })
   391→    const steps = movePath(g0, move)
   392→    for (const step of steps) {
   393→      if (gen !== loopGen) return
   394→      await sleep(step.location === 'finished' ? 260 : 130)
   395→      if (gen !== loopGen) return
   396→      set({ game: patchToken(g0, move, step) })
   397→      sfx.step()
   398→    }
   399→    if (gen !== loopGen) return
   400→    await sleep(120)
   401→    const result = applyMove(g0, move)
   402→    if (move.captures.length > 0) {
   403→      sfx.capture()
   404→      pushFloat(`Captured ${move.captures.length} token${move.captures.length > 1 ? 's' : ''}!`, 'capture')
   405→    }
   406→    if (move.finishes) sfx.finish()
   407→    set({ game: result, animating: false, diceValues: [] })
   408→    if (result.phase === 'gameover' && result.winner) {
   409→      finishGame(result)
   410→      return
   411→    }
   412→    scheduleNext()
   413→  }
   414→
   415→  async function botRoll(gen: number) {
   416→    if (gen !== loopGen) return
   417→    const g = get().game
   418→    if (!g || g.phase !== 'roll') return
   419→    set({ canRoll: false, diceRolling: true })
   420→    sfx.dice()
   421→    await sleep(650)
   422→    if (gen !== loopGen) return
   423→    const { state, rolled, forfeited } = rollDice(g)
   424→    set({ game: state, diceValues: rolled, diceRolling: false, selectedDie: 0 })
   425→    if (rolled[0] === 6 || (state.mode === 'master' && rolled.every((d) => d === 6))) sfx.six()
   426→    if (forfeited) {
   427→      pushFloat(`Three sixes! ${g.players[g.turn].name} loses the turn`, 'forfeit')
   428→      const passed = passTurn(state)
   429→      set({ game: passed })
   430→      schedule(1100, () => scheduleNext())
   431→      return
   432→    }
   433→    await sleep(250)
   434→    if (gen !== loopGen) return
   435→    scheduleNext()
   436→  }
   437→
   438→  async function botMove(gen: number) {
   439→    if (gen !== loopGen) return
   440→    const g = get().game
   441→    if (!g || g.phase !== 'move') return
   442→    const move = botChooseMove(g)
   443→    if (!move) {
   444→      const passed = passTurn(g)
   445→      set({ game: passed })
   446→      schedule(900, () => scheduleNext())
   447→      return
   448→    }
   449→    set({ selectedDie: move.dieIndex, diceValues: [...g.dice] })
   450→    await animateAndApply(g, move)
   451→  }

   453→  // ---- online event wiring ----
   454→
   455→  function wireOnline() {
   456→    const sock = getSocket()
   457→    sock.off('connect')
   458→    sock.off('disconnect')
   459→    sock.on('connect', () => {
   460→      set((s) => ({ onlineStatus: s.lobby ? 'lobby' : 'connected', onlineError: '' }))
   461→    })
   462→    sock.on('disconnect', () => {
   463→      set((s) => (s.onlineStatus === 'playing' ? { onlineStatus: 'playing' } : { onlineStatus: 'idle' }))
   464→    })
   465→    online.on('lobby:created', ({ code }: { code: string }) => {
   466→      set({ onlineStatus: 'lobby', view: 'lobby' })
   467→    })
   468→    online.on('lobby:update', ({ room }: { room: LobbyInfo }) => {
   469→      set((s) => ({
   470→        onlineStatus: 'lobby',
   471→        lobby: room,
   472→        onlineError: '',
   473→        view: s.view === 'game' ? 'game' : 'lobby',
   474→      }))
   475→    })
   476→    online.on('lobby:error', ({ message }: { message: string }) => {
   477→      set({ onlineError: message })
   478→    })
   479→    online.on('game:started', ({ state, yourColor, roomCode, stake }: { state: GameState; yourColor: Color; roomCode: string; stake: number }) => {
   480→      loopGen++
   481→      cancelSchedule()
   482→      const pot = stake * state.players.length
   483→      const me = get().profile
   484→      set({
   485→        view: 'game',
   486→        session: { mode: state.mode, stake, online: true, pot, youColor: yourColor, roomCode },
   487→        game: state,
   488→        onlineStatus: 'playing',
   489→        winData: null,
   490→        animating: false,
   491→        diceRolling: false,
   492→        diceValues: [],
   493→        canRoll: false,
   494→        movableTokenIds: [],
   495→        selectedDie: 0,
   496→        chat: [],
   497→        floatMsgs: [],
   498→        rematchVotes: [],
   499→        lobby: null,
   500→        turnDeadline: null,
   501→      })
   502→      get().addCoins(-stake) // pay entry
   503→      sfx.coin()
   504→      void me
   505→      // drive my turn if it starts now
   506→      setTimeout(() => scheduleNextOnline(), 400)
   507→    })
   508→    online.on('game:rolled', ({ state, rolled, forfeited }: { state: GameState; rolled: number[]; forfeited: boolean }) => {
   509→      const gen = ++loopGen
   510→      cancelSchedule()
   511→      set({ game: state, diceValues: rolled, diceRolling: false, canRoll: false, selectedDie: 0 })
   512→      if (rolled[0] === 6) sfx.six()
   513→      if (forfeited) {
   514→        pushFloat('Three sixes! Turn forfeited', 'forfeit')
   515→      }
   516→      setTimeout(() => scheduleNextOnline(), 300)
   517→      void gen
   518→    })
   519→    online.on('game:moved', ({ state, move }: { state: GameState; move: OnlineMove }) => {
   520→      // replay animation from previous visible state, then snap to authoritative state
   521→      const g0 = get().game
   522→      const gen = ++loopGen
   523→      cancelSchedule()
   524→      if (g0 && move) {
   525→        void (async () => {
   526→          set({ animating: true, movableTokenIds: [] })
   527→          const steps = movePathForOnline(g0, move)
   528→          for (const step of steps) {
   529→            if (gen !== loopGen) return
   530→            await sleep(step.location === 'finished' ? 260 : 130)
   531→            if (gen !== loopGen) return
   532→            set({ game: patchTokenOnline(g0, move, step) })
   533→            sfx.step()
   534→          }
   535→          if (gen !== loopGen) return
   536→          if (move.captures.length > 0) {
   537→            sfx.capture()
   538→            pushFloat('Capture!', 'capture')
   539→          }
   540→          if (move.finishes) sfx.finish()
   541→          set({ game: state, animating: false, diceValues: [] })
   542→          setTimeout(() => scheduleNextOnline(), 250)
   543→        })()
   544→      } else {
   545→        set({ game: state, animating: false })
   546→        setTimeout(() => scheduleNextOnline(), 250)
   547→      }
   548→    })
   549→    online.on('game:passed', ({ state, reason }: { state: GameState; reason?: string }) => {
   550→      set({ game: state, diceValues: [] })
   551→      pushFloat(reason === 'forfeit' ? 'Three sixes! Turn forfeited' : 'No possible move', reason === 'forfeit' ? 'forfeit' : 'pass')
   552→      sfx.pass()
   553→      setTimeout(() => scheduleNextOnline(), 300)
   554→    })
   555→    online.on('game:state', ({ state }: { state: GameState }) => {
   556→      // full authoritative sync (bot takeover, reconnect)
   557→      const gen = ++loopGen
   558→      cancelSchedule()
   559→      set({ game: state, animating: false, diceRolling: false, diceValues: [], movableTokenIds: [] })
   560→      void gen
   561→      setTimeout(() => scheduleNextOnline(), 300)
   562→    })
   563→    online.on('game:over', ({ winner, winnerName, pot }: { winner: Color; winnerName: string; pot: number }) => {
   564→      const session = get().session
   565→      if (!session) return
   566→      const youWon = winner === session.youColor
   567→      if (youWon) {
   568→        get().addCoins(pot)
   569→        sfx.win()
   570→        set((s) => ({ profile: { ...s.profile, wins: s.profile.wins + 1, gems: s.profile.gems + 2 } }))
   571→      } else {
   572→        sfx.lose()
   573→        set((s) => ({ profile: { ...s.profile, losses: s.profile.losses + 1 } }))
   574→      }
   575→      saveProfile(get().profile)
   576→      set({
   577→        winData: { winner, winnerName, pot, youWon, coinDelta: youWon ? pot : 0 },
   578→        canRoll: false,
   579→        movableTokenIds: [],
   580→        animating: false,
   581→        turnDeadline: null,
   582→      })
   583→    })
   584→    online.on('game:chat', (msg: { color: Color; name: string; text: string }) => {
   585→      set((s) => ({ chat: [...s.chat, { id: msgId++, color: msg.color, name: msg.name, text: msg.text, ts: Date.now() }].slice(-50) }))
   586→    })
   587→    online.on('game:emote', ({ color, emoji }: { color: Color; emoji: string }) => {
   588→      sfx.emote()

   603→  /** Online equivalent of scheduleNext: only enables local input; bots are server-driven. */
   604→  function scheduleNextOnline() {
   605→    const s = get()
   606→    if (!s.game || s.game.phase === 'gameover' || !s.session?.online) return
   607→    const g = s.game
   608→    const cur = g.players[g.turn]
   609→    const isMe = cur.color === s.session.youColor && !cur.isBot
   610→    if (g.phase === 'roll') {
   611→      if (isMe) {
   612→        set({ canRoll: true })
   613→        startTurnTimer()
   614→      } else {
   615→        set({ canRoll: false })
   616→      }
   617→      return
   618→    }
   619→    if (g.phase === 'move') {
   620→      if (isMe) {
   621→        set({ canRoll: false })
   622→        computeMovable()
   623→      } else {
   624→        set({ movableTokenIds: [] })
   625→      }
   626→    }
   627→  }

   629→  function movePathForOnline(g0: GameState, move: OnlineMove) {
   630→    // find moving token in current display state (turn might differ from server snapshot ordering)
   631→    const color = move.color
   632→    const token = g0.players.find((p) => p.color === color)?.tokens[move.tokenId]
   633→    if (!token) return []
   634→    const die = move.die
   635→    const from = token.location === 'yard' ? -1 : token.location === 'home' ? 50 + token.homeStep : token.location === 'finished' ? 56 : token.progress
   636→    const to = token.location === 'yard' ? 0 : Math.min(from + die, 56)
   637→    const steps: Array<{ location: 'track' | 'home' | 'finished'; trackIdx: number; homeStep: number }> = []
   638→    for (let p = from + 1; p <= to; p++) {
   639→      if (p <= 50) steps.push({ location: 'track', trackIdx: trackIdxFor(color, p), homeStep: 0 })
   640→      else if (p < 56) steps.push({ location: 'home', trackIdx: -1, homeStep: p - 50 })
   641→      else steps.push({ location: 'finished', trackIdx: -1, homeStep: 0 })
   642→    }
   643→    return steps
   644→  }
